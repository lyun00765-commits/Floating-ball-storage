## 准备

### 造测试用的球

在**酒馆主页面**控制台执行：

```js
// 造 n 个「应该被收纳」的悬浮球
window.__mkBall = (n = 1, size = 48) => {
  for (let i = 1; i <= n; i++) {
    const d = document.createElement('div')
    d.id = 'test-ball-' + Date.now() + '-' + i
    d.className = 'test-floating-ball'
    d.style.cssText = `position:fixed;right:${20 + i * 60}px;bottom:120px;width:${size}px;height:${size}px;`
      + 'border-radius:50%;background:#4a9eff;color:#fff;display:flex;align-items:center;'
      + 'justify-content:center;cursor:pointer;z-index:9999'
    d.textContent = '球'
    document.body.appendChild(d)
  }
  console.log('现有测试球:', document.querySelectorAll('.test-floating-ball').length)
}

// 造一个「不该被收纳」的关闭按钮
window.__mkCloseBtn = () => {
  const d = document.createElement('div')
  d.className = 'close-btn fa-times'
  d.style.cssText = 'position:fixed;left:20px;top:20px;width:32px;height:32px;border-radius:50%;'
    + 'background:#f66;color:#fff;display:flex;align-items:center;justify-content:center;cursor:pointer;z-index:9999'
  d.textContent = '×'
  document.body.appendChild(d)
}

// 清理
window.__cleanBalls = () =>
  document.querySelectorAll('.test-floating-ball,.close-btn').forEach(e => e.remove())
```

> 这些球能被识别的原因：`fixed`(+2) + 圆形(+2) + `cursor:pointer`(+1) = 5 分，超过阈值 4。
